# HIG Gaps (Stage 2) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Close the HIG gaps found in review: a confirmed "Remove from Backlog", a VoiceOver-friendly rating, layouts that reflow at accessibility text sizes, and a `…` menu on Home rows. Swipe-to-remove was dropped: the row menu covers it (spec §3.5).

**Architecture:** Pure decisions (copy, spoken labels, the accessibility-size threshold) live in plain `.ts` modules with vitest tests. The Home row menu is an `@expo/ui` SwiftUI `Menu` in a small `Host`, passed into `GameRow` as a trailing `menu` slot by a `BacklogGameRow` that owns the per-row mutations. No local native modules, no keyboard shortcuts (spec §3.6).

**Tech Stack:** Expo SDK 57, `@expo/ui` SwiftUI (`Menu`, `Picker`, `Button`, `Divider`), vitest.

**Spec:** `docs/superpowers/specs/2026-10-12-ipad-duo-adaptive-layout-design.md` §3 (Stage 2). Builds on Stage 1, on the same branch.

## Global Constraints

- Branch `feat/mobile-1.2`, which already holds Stage 1; all three stages merge together as mobile 1.2.0.
- Menus only on Home rows and cards; Search/Explore/Similar/Share lists are unchanged (spec §3.4 — summaries carry no rating, and the upsert would wipe it).
- No `apps/mobile/modules/`, no new native dependencies, no keyboard shortcuts (spec §3.6).
- Navigation is unchanged: rows keep `onPress` → `router.push` within their tab.
- The `…` button is a sibling of the row's `Pressable`, never inside it.
- Every status change goes through `useUpsertBacklogEntry` (haptic, §2.6a) and `wouldExceedSlots` (paywall gate).
- Every removal goes through `confirmRemove` — toolbar and row menu alike.
- `isAccessibilitySize` threshold: `fontScale >= 1.6`.
- Anything under `apps/mobile/test/` must not import `react-native`, `@expo/ui` or `expo-*`.
- Colours via `PlatformColor`; the only hex literal stays `Brand.tint`.
- Lint `--max-warnings 0`, Prettier, conventional commits.

## Review Focus

1. **Tapping the `…` button opens only the menu**, never the game as well; tapping anywhere else on the row or card opens the game. Checked manually in Task 4 and Task 5.
2. **A status change from the menu keeps the rating** and is blocked by the paywall gate exactly as `EntryActions` is. Pinned by passing `entry.rating` in Task 4's code and by the shared `wouldExceedSlots`; checked manually in Task 5.
3. **Cancelling a removal does nothing** — from the toolbar and the menu. Pinned by `removeConfirmCopy` tests (Task 1) and manual checks.
4. **VoiceOver never reads "black star"** — rows read "Playing, rated 9 out of 10", the rating button reads "Rating, 9 out of 10", the `…` button reads "More, <game>". Pinned by `format.test.ts` (Task 2).
5. **Scrolling a long backlog stays smooth** with one SwiftUI `Host` per row. Checked with 200 games on a physical iPhone (Task 5); fallback in spec §3.4.

import { Alert } from "react-native";

import { removeConfirmCopy } from "@/features/game/remove-confirm";

/**
 * The one path to removing a game — toolbar and row menu both come
 * through here, so they all confirm. Undo was rejected: iOS has no system
 * toast to host it.
 */
export function confirmRemove(gameName: string, onConfirm: () => void): void {
  const copy = removeConfirmCopy(gameName);

  Alert.alert(copy.title, copy.message, [
    { text: copy.cancel, style: "cancel" },
    { text: copy.confirm, style: "destructive", onPress: onConfirm },
  ]);
}
```

export function ratingButtonLabel(rating: number | null): string {
  return rating === null ? "Rate" : String(rating);
}

export function ratingButtonSymbol(rating: number | null): SFSymbol {
  return rating === null ? "star" : "star.fill";
}

/** For VoiceOver: the visible label is a bare number beside a star. */
export function ratingSpoken(rating: number | null): string {
  return rating === null ? "Rate this game" : `Rating, ${rating} out of ${RATING_MAX}`;
}

/** `rowSubtitle` without the `★`, which VoiceOver reads as "black star". */
export function rowSubtitleSpoken(entry: { status: BacklogStatus; rating: number | null }): string {
  return entry.rating === null
    ? statusLabel(entry.status)
    : `${statusLabel(entry.status)}, rated ${entry.rating} out of ${RATING_MAX}`;
}

/** Just under iOS's first accessibility size (AX1 reports ~1.65); every
 * standard size, up to xxxLarge at ~1.35, stays below it. */
export const ACCESSIBILITY_FONT_SCALE = 1.6;

/** The HIG asks horizontal layouts to stack at these sizes so text gets the
 * full width. */
export function isAccessibilitySize(fontScale: number): boolean {
  return Number.isFinite(fontScale) && fontScale >= ACCESSIBILITY_FONT_SCALE;
}
```

const card = variant === "card";
const body = (
  <Pressable
    onPress={() => onPress(id)}
    style={({ pressed }) => [
      card ? styles.card : styles.row,
      menu !== undefined && styles.withMenu,
      pressed && (card ? styles.cardPressed : styles.pressed),

    accessibilityRole="button"
    accessibilityLabel={/* unchanged from Task 2 */}

    ) : null}
  </Pressable>
);

if (menu === undefined) return body;

// A sibling on top, not a child: a tap on the menu must not also open the game.

  <View style={card ? styles.cardSlot : undefined}>

- [ ] **Step 3: `backlog-row-menu.tsx`:**

```tsx
import { Button, Divider, Host, Menu, Picker, Text } from "@expo/ui/swift-ui";
import {
  accessibilityLabel,
  buttonStyle,
  foregroundStyle,
  labelStyle,
  pickerStyle,
  tag,
} from "@expo/ui/swift-ui/modifiers";
import type { BacklogStatus } from "@repo/contracts";
import { PlatformColor, StyleSheet } from "react-native";

import { STATUS_ORDER } from "@/features/backlog/sections";
import { statusLabel, statusSymbol } from "@/features/game/format";
import { Brand } from "@/theme";

/**
 * The `…` on a Home row. A visible button rather than a long-press menu: it is
 * found without guessing, and touch, VoiceOver and pointer all reach it the same
 * way (spec §3.4). SwiftUI's own `Menu`, so the pointer effect comes free.
 */
export function BacklogRowMenu({
  name,
  status,
  onStatus,
  onShare,
  onRemove,
}: {
  name: string;
  status: BacklogStatus;
  onStatus: (next: BacklogStatus) => void;
  onShare: () => void;
  onRemove: () => void;
}) {
  return (
    <Host style={styles.host} seedColor={Brand.tint}>
      <Menu
        label="More"
        systemImage="ellipsis.circle"
        modifiers={[
          labelStyle("iconOnly"),
          buttonStyle("borderless"),
          foregroundStyle(PlatformColor("secondaryLabel")),
          accessibilityLabel(`More, ${name}`),
        ]}
      >
        <Menu label="Status" systemImage={statusSymbol(status)}>
          <Picker
            selection={status}
            onSelectionChange={(selection) => onStatus(selection as BacklogStatus)}
            modifiers={[pickerStyle("inline")]}
          >
            {STATUS_ORDER.map((value) => (
              <Text key={value} modifiers={[tag(value)]}>
                {statusLabel(value)}
              </Text>
            ))}
          </Picker>
        </Menu>
        <Button label="Share" systemImage="square.and.arrow.up" onPress={onShare} />
        <Divider />
        <Button
          label="Remove from Backlog…"
          systemImage="trash"
          role="destructive"
          onPress={onRemove}
        />
      </Menu>
    </Host>
  );
}

const styles = StyleSheet.create({
  host: { width: 44, height: 44 },
});
```

- [ ] **Step 4: `backlog-game-row.tsx`:**

```tsx
import type { BacklogListItemWire, BacklogStatus } from "@repo/contracts";
import { useRouter } from "expo-router";

import {
  useBacklogStats,
  useDeleteBacklogEntry,
  usePremium,
  useUpsertBacklogEntry,
} from "@/api/hooks";
import { GameRow, type GameRowVariant } from "@/components/game-row";
import { BacklogRowMenu } from "@/features/backlog/backlog-row-menu";
import { activeSlotsUsed } from "@/features/backlog/slots";
import { confirmRemove } from "@/features/game/confirm-remove";
import { rowSubtitle, rowSubtitleSpoken } from "@/features/game/format";
import { shareGame } from "@/features/game/share-game";
import { wouldExceedSlots } from "@/features/paywall/should-offer-paywall";

/**
 * A Home row with the backlog's own menu. Home is the one list that knows each
 * game's entry, rating included — so it is the one list that may change status
 * without wiping the rating (spec §3.4).
 */
export function BacklogGameRow({
  entry,
  variant,
  onPress,
}: {
  entry: BacklogListItemWire;
  variant: GameRowVariant;
  onPress: (id: number) => void;
}) {
  const upsert = useUpsertBacklogEntry(entry.gameId);
  const remove = useDeleteBacklogEntry(entry.gameId);
  const premium = usePremium();
  const stats = useBacklogStats();
  const router = useRouter();
  const activeCount = stats.data === undefined ? undefined : activeSlotsUsed(stats.data);

  const setStatus = (next: BacklogStatus) => {
    if (next === entry.status) return;

    if (wouldExceedSlots({ premium, activeCount, from: entry.status, to: next })) {
      router.push("/paywall");
      return;
    }

    upsert.mutate({ status: next, rating: entry.rating });
  };

  const askRemove = () => confirmRemove(entry.game.name, () => remove.mutate());

  return (
    <GameRow
      id={entry.gameId}
      title={entry.game.name}
      subtitle={rowSubtitle(entry)}
      spokenSubtitle={rowSubtitleSpoken(entry)}
      coverImageId={entry.game.coverImageId}
      variant={variant}
      onPress={onPress}
      menu={
        <BacklogRowMenu
          name={entry.game.name}
          status={entry.status}
          onStatus={setStatus}
          onShare={() => shareGame(entry.game)}
          onRemove={askRemove}
        />
      }
    />
  );
}
```

`backlog-screen.tsx` `renderItem`: the `game(entry, variant)` helper returns `<BacklogGameRow key={entry.gameId} entry={entry} variant={variant} onPress={openGame} />`; drop the now-unused `GameRow` value import (keep `type GameRowVariant`) and `rowSubtitle`.
